package main

import "fmt"

type Service struct {
	Host string
	Port int
}

func InspectVariable(val any) (fieldFormat, syntaxFormat string) {
	fieldFormat = fmt.Sprintf("%+v", val)
	syntaxFormat = fmt.Sprintf("%#v", val)
	return fieldFormat, syntaxFormat
}

func main() {
	s := Service{Host: "127.0.0.1", Port: 9000}
	f, _ := InspectVariable(s)
	fmt.Printf("Field format: %s\n", f)
}

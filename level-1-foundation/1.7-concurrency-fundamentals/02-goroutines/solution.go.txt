package main

import (
	"fmt"
)

// LaunchWorker meluncurkan goroutine yang mengirim status selesai ke channel out.
func LaunchWorker(id int, out chan<- string) {
	go func() {
		out <- fmt.Sprintf("Worker %d done", id)
	}()
}

func main() {
	out := make(chan string)
	LaunchWorker(1, out)
	msg := <-out
	fmt.Println(msg)
}

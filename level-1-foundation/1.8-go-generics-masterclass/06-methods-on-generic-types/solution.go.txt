package main

import "fmt"

// Stack adalah struktur data tumpukan Last-In-First-Out berpengetikan generik.
type Stack[T any] struct {
	items []T
}

func (s *Stack[T]) Push(val T) {
	s.items = append(s.items, val)
}

func (s *Stack[T]) Pop() (T, bool) {
	if len(s.items) == 0 {
		var zero T
		return zero, false
	}
	idx := len(s.items) - 1
	item := s.items[idx]
	s.items = s.items[:idx]
	return item, true
}

func (s *Stack[T]) Len() int {
	return len(s.items)
}

func main() {
	var stack Stack[int]
	stack.Push(1)
	stack.Push(2)
	stack.Push(3)

	val, _ := stack.Pop()
	fmt.Printf("Pop: %d, Len: %d\n", val, stack.Len())
}

package main

import "fmt"

func FormatData(val any) string {
	switch v := val.(type) {
	case string:
		return fmt.Sprintf("STR: %s", v)
	case int:
		return fmt.Sprintf("INT: %d", v*2)
	case bool:
		if v {
			return "BOOL: YES"
		}
		return "BOOL: NO"
	default:
		return "UNKNOWN"
	}
}

func main() {
	fmt.Println(FormatData("golang"))
	fmt.Println(FormatData(10))
	fmt.Println(FormatData(true))
	fmt.Println(FormatData(false))
	fmt.Println(FormatData(3.14))
}

package main

import (
	"fmt"
	"strings"
)

// isAllowedSlugChar memeriksa apakah rune adalah alfanumerik, spasi, atau dash (-).
func isAllowedSlugChar(r rune) bool {
	isAlphaNumeric := (r >= 'a' && r <= 'z') || (r >= 'A' && r <= 'Z') || (r >= '0' && r <= '9')
	isSep := r == ' ' || r == '-'
	return isAlphaNumeric || isSep
}

// countInvalidChars menghitung banyaknya karakter yang tidak diizinkan pada text.
func countInvalidChars(text string) int {
	count := 0
	for _, r := range text {
		if !isAllowedSlugChar(r) {
			count++
		}
	}
	return count
}

// cleanText membuang karakter tidak valid, mengubah ke huruf kecil, dan merapikan spasi di ujung teks.
func cleanText(text string) string {
	var filtered string
	for _, r := range text {
		if isAllowedSlugChar(r) {
			filtered += string(r)
		}
	}
	return strings.TrimSpace(strings.ToLower(filtered))
}

// transformToSlug mengubah teks menjadi format slug URL (memadatkan pemisah menjadi satu dash).
func transformToSlug(text string) string {
	cleaned := cleanText(text)
	var result string
	inSeparator := false

	for _, r := range cleaned {
		if r == ' ' || r == '-' {
			if !inSeparator && len(result) > 0 {
				result += "-"
				inSeparator = true
			}
		} else {
			result += string(r)
			inSeparator = false
		}
	}
	return strings.Trim(result, "-")
}

func main() {
	input := "  Go 101: Belajar #1!  "
	slug := transformToSlug(input)
	fmt.Printf("Input: %q\n", input)
	fmt.Printf("Slug:  %q\n", slug)
}

package main

import (
	"errors"
	"fmt"
)

func divide(a, b float64) (float64, error) {
	if b == 0 {
		return 0, errors.New("cannot divide by zero")
	}
	return a / b, nil
}

func main() {
	res1, err1 := divide(10, 2)
	if err1 != nil {
		fmt.Println(err1)
	} else {
		fmt.Println(res1)
	}

	res2, err2 := divide(10, 0)
	if err2 != nil {
		fmt.Println(err2)
	} else {
		fmt.Println(res2)
	}
}
